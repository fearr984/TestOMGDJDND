-- REMOTES  (Rivals path; waits up to 8s each)
-- ════════════════════════════════════════════════════════════════
local Remotes = {}
local function tryGet(...)
    local path = {...}
    local ok, res = pcall(function()
        local node = cloneref_(ReplicatedStorage)
        for _, p in ipairs(path) do
            node = node:WaitForChild(p, 8)
            if not node then return nil end
            node = cloneref_(node)
        end
        return node
    end)
    return ok and res or nil
end

local function fireRemote(remote, ...)
    if not remote then return end
    pcall(function() remote:FireServer(...) end)
end

-- ════════════════════════════════════════════════════════════════
-- BACKTRACK BUFFER  (Kicia-style: ring buffer of past root CFrames)
-- ════════════════════════════════════════════════════════════════
local BacktrackBuffer = {}   -- [player] = { {cf, t}, ... }
local BT_MAX_MS = 400

local function btPush(p)
    local char = p.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local now = os.clock()
    local buf = BacktrackBuffer[p]
    if not buf then buf = {}; BacktrackBuffer[p] = buf end
    table.insert(buf, { cf = root.CFrame, t = now })
    -- prune old entries
    local cutoff = now - (BT_MAX_MS / 1000)
    while buf[1] and buf[1].t < cutoff do table.remove(buf, 1) end
end

local function btBest(p)
    -- return the CFrame entry within the configured window, prefer furthest back
    local buf = BacktrackBuffer[p]
    if not buf or #buf == 0 then return nil end
    local now = os.clock()
    local window = Cfg.Ragebot.BacktrackMs / 1000
    for i = #buf, 1, -1 do
        if (now - buf[i].t) <= window then return buf[i].cf end
    end
    return nil
end

RunService.Heartbeat:Connect(function()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp then btPush(p) end
    end
end)

-- ════════════════════════════════════════════════════════════════
-- TARGETING UTILS
-- ════════════════════════════════════════════════════════════════
local function getLocalRoot()
    local c = lp.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function getAimPart(char, partName)
    return char:FindFirstChild(partName) or char:FindFirstChild("HumanoidRootPart")
end

local function screenDist(worldPos)
    local sp, onScreen = camera:WorldToViewportPoint(worldPos)
    if not onScreen then return math.huge end
    local center = camera.ViewportSize / 2
    return (Vector2.new(sp.X, sp.Y) - center).Magnitude
end

local function isVisible(part)
    local origin = camera.CFrame.Position
    local dir    = (part.Position - origin)
    local result = workspace:Raycast(origin, dir, RaycastParams.new())
    if not result then return true end
    -- visible if the first hit IS the target part or its parent character
    local hit = result.Instance
    return hit == part or (hit and hit:IsDescendantOf(part.Parent))
end

-- velocity-predicted position
local function predictedPos(part)
    local vel = part.AssemblyLinearVelocity
    return part.Position + vel * Cfg.Aimbot.Prediction
end

-- backstab offset: position behind the target's root
local function backstabPos(char)
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return nil end
    return root.CFrame * CFrame.new(0, 0, 1.5)  -- 1.5 studs behind
end

local function getBestTarget(fovRadius, partName)
    local best, bestDist = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and p.Character then
            local hum  = p.Character:FindFirstChild("Humanoid")
            local part = getAimPart(p.Character, partName or "Head")
            if hum and hum.Health > 0 and part then
                if Cfg.Ragebot.Priority == "Visible" and not isVisible(part) then continue end
                local d = screenDist(predictedPos(part))
                if d < fovRadius and d < bestDist then
                    bestDist = d; best = part
                end
            end
        end
    end
    return best
end

local function getClosestPlayer()
    local best, bestDist = nil, math.huge
    local localRoot = getLocalRoot()
    if not localRoot then return nil end
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and p.Character then
            local hum  = p.Character:FindFirstChild("Humanoid")
            local root = p.Character:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health > 0 and root then
                local d = (localRoot.Position - root.Position).Magnitude
                if d < bestDist then bestDist = d; best = p end
            end
        end
    end
    return best
end

local function getLowHPPlayer()
    local best, bestHP = nil, math.huge
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and p.Character then
            local hum = p.Character:FindFirstChild("Humanoid")
            if hum and hum.Health > 0 and hum.Health < bestHP then
                bestHP = hum.Health; best = p
            end
        end
    end
    return best
end

local function getVisiblePlayer()
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= lp and p.Character then
            local hum  = p.Character:FindFirstChild("Humanoid")
            local root = p.Character:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health > 0 and root and isVisible(root) then
                return p
            end
        end
    end
    return nil
end

local function pickTarget()
    if Cfg.Ragebot.Priority == "LowHP"   then return getLowHPPlayer()
    elseif Cfg.Ragebot.Priority == "Visible" then return getVisiblePlayer() or getClosestPlayer()
    else return getClosestPlayer() end
end

-- ════════════════════════════════════════════════════════════════
-- AUTO WEAPON SWITCH  (Kicia pattern: scan character tools)
-- cycles to the first usable tool that isn't currently held
-- ════════════════════════════════════════════════════════════════
local function autoWeaponSwitch(char)
    if not char then return end
    local hum = char:FindFirstChild("Humanoid")
    if not hum then return end
    -- Rivals stores weapons in character; equip first tool that has a Handle
    for _, tool in ipairs(char:GetChildren()) do
        if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
            -- already equipped if it's in the character directly
            -- re-equip via humanoid to force the server to register it
            pcall(function() hum:EquipTool(tool) end)
            return
        end
    end
    -- fallback: check backpack
    local bp = lp:FindFirstChild("Backpack")
    if bp then
        for _, tool in ipairs(bp:GetChildren()) do
            if tool:IsA("Tool") and tool:FindFirstChild("Handle") then
                pcall(function() hum:EquipTool(tool) end)
                return
            end
        end
    end
end

-- ════════════════════════════════════════════════════════════════
-- CAMERA MANIPULATION  (Kicia-native, no UE remote)
-- fires UpdateCameraRotation with a spoofed CFrame pointing at target
-- ════════════════════════════════════════════════════════════════
local function applyManip(targetPart)
    if not targetPart then return end
    local localChar = lp.Character
    if not localChar then return end
    local head = localChar:FindFirstChild("Head") or localChar:FindFirstChild("HumanoidRootPart")
    if not head then return end
    local aimPos = Cfg.Ragebot.Backstab and backstabPos(targetPart.Parent)
                   and backstabPos(targetPart.Parent).Position
                   or predictedPos(targetPart)
    local dir     = (aimPos - head.Position).Unit
    local manipCF = CFrame.new(head.Position, head.Position + dir)
    if Remotes.CamRot then fireRemote(Remotes.CamRot, manipCF) end
end

-- ════════════════════════════════════════════════════════════════
-- DESYNC  (Kicia pattern: SetServerCFrame to far coord each frame)
-- implemented via char CFrame teleport while Desync is active
-- ════════════════════════════════════════════════════════════════
local desyncActive = false
local DESYNC_LIMIT = 2147483646
local DESYNC_DEAD  = 1147483646

local function farCoord()
    local m = math.random(DESYNC_DEAD + 1, DESYNC_LIMIT)
    return math.random(0, 1) == 0 and -m or m
end
local function farVec() return Vector3.new(farCoord(), farCoord(), farCoord()) end

local function applyDesync()
    if not Cfg.Ragebot.Desync then desyncActive = false; return end
    local localChar = lp.Character
    local root = localChar and localChar:FindFirstChild("HumanoidRootPart")
    if not root then return end
    -- fire a camera-rot remote with a far CFrame to desync server hitbox
    if Remotes.CamRot then
        pcall(function()
            Remotes.CamRot:FireServer(CFrame.new(farVec()) * CFrame.Angles(math.pi, math.pi, math.pi))
        end)
    end
    desyncActive = true
end

local ESPObjects = {}

local function removeESP(p)
    if ESPObjects[p] then
        for _, obj in pairs(ESPObjects[p]) do
            pcall(function() obj:Remove() end)
        end
        ESPObjects[p] = nil
    end
end

local function makeESP(p)
    if not p.Character then return end
    removeESP(p)
    local obj = {}

    local box = Drawing.new("Square")
    box.Visible = false; box.Color = Cfg.Visuals.BoxColor
    box.Thickness = 1.5; box.Filled = false
    obj.box = box

    local nametag = Drawing.new("Text")
    nametag.Visible = false; nametag.Color = Color3.fromRGB(220, 230, 255)
    nametag.Size = 13; nametag.Center = true; nametag.Outline = true
    obj.name = nametag

    local hbar = Drawing.new("Square")
    hbar.Visible = false; hbar.Color = Color3.fromRGB(80, 255, 60)
    hbar.Filled = true
    obj.hbar = hbar

    local snap = Drawing.new("Line")
    snap.Visible = false; snap.Color = Cfg.Visuals.TracerColor
    snap.Thickness = 1
    obj.snap = snap

    ESPObjects[p] = obj
end

local function updateESP()
    for _, p in ipairs(Players:GetPlayers()) do
        if p == lp then continue end
        if not ESPObjects[p] then makeESP(p) end
        local obj  = ESPObjects[p]
        if not obj then continue end
        local char = p.Character
        local hum  = char and char:FindFirstChild("Humanoid")
        local root = char and char:FindFirstChild("HumanoidRootPart")
        if not char or not hum or not root or hum.Health <= 0 then
            for _, o in pairs(obj) do o.Visible = false end
            continue
        end
        local topSP, onTop = camera:WorldToViewportPoint(root.Position + Vector3.new(0, 3.2, 0))
        local botSP, onBot = camera:WorldToViewportPoint(root.Position - Vector3.new(0, 2.8, 0))
        if not onTop or not onBot then
            for _, o in pairs(obj) do o.Visible = false end
            continue
        end
        local height = math.abs(topSP.Y - botSP.Y)
        local width  = height * 0.5
        local cx, cy = topSP.X, topSP.Y

        if Cfg.Visuals.ESP and obj.box then
            obj.box.Position = Vector2.new(cx - width/2, cy)
            obj.box.Size     = Vector2.new(width, height)
            obj.box.Visible  = true
        elseif obj.box then obj.box.Visible = false end

        if Cfg.Visuals.NameTags and obj.name then
            obj.name.Position = Vector2.new(cx, cy - 14)
            obj.name.Text     = p.Name .. " [" .. math.floor(hum.Health) .. "]"
            obj.name.Visible  = true
        elseif obj.name then obj.name.Visible = false end

        if Cfg.Visuals.HealthBars and obj.hbar then
            local hp = hum.Health / hum.MaxHealth
            obj.hbar.Position = Vector2.new(cx - width/2 - 5, cy + height * (1 - hp))
            obj.hbar.Size     = Vector2.new(3, height * hp)
            obj.hbar.Visible  = true
        elseif obj.hbar then obj.hbar.Visible = false end

        if Cfg.Visuals.Snaplines and obj.snap then
            obj.snap.From    = camera.ViewportSize / 2
            obj.snap.To      = Vector2.new(topSP.X, topSP.Y)
            obj.snap.Visible = true
        elseif obj.snap then obj.snap.Visible = false end
    end
end

local ChamObjects = {}

local function applyChams(p)
    if not p.Character then return end
    ChamObjects[p] = ChamObjects[p] or {}
    for _, part in ipairs(p.Character:GetDescendants()) do
        if part:IsA("BasePart") and not ChamObjects[p][part] then
            local hl = Instance.new("SelectionBox")
            hl.Adornee           = part
            hl.Color3            = Color3.fromRGB(80, 255, 160)
            hl.LineThickness     = 0.03
            hl.SurfaceTransparency = 0.6
            hl.Parent            = part
            ChamObjects[p][part] = hl
        end
    end
end

local function removeChams(p)
    if ChamObjects[p] then
        for _, hl in pairs(ChamObjects[p]) do
            pcall(function() hl:Destroy() end)
        end
        ChamObjects[p] = nil
    end
end

local function setFly(state)
    local char = lp.Character
    if not char then return end
    local root = char:FindFirstChild("HumanoidRootPart")
    if not root then return end
    if state then
        local bg = Instance.new("BodyGyro")
        bg.MaxTorque = Vector3.new(1e9, 1e9, 1e9)
        bg.D = 100; bg.P = 1e4; bg.Parent = root
        local bv = Instance.new("BodyVelocity")

        if flyBody then
            pcall(function() flyBody.bg:Destroy() end)
            pcall(function() flyBody.bv:Destroy() end)
            flyBody = nil
        end
    end
end

-- ════════════════════════════════════════════════════════════════
-- RAGEBOT STATE
-- ════════════════════════════════════════════════════════════════
local RageState = { lastFire = 0, lastTrigger = 0 }

-- ════════════════════════════════════════════════════════════════
-- HEARTBEAT — all per-frame logic
-- ════════════════════════════════════════════════════════════════
RunService.Heartbeat:Connect(function(dt)
    local now = os.clock()

    -- ── AIMBOT ───────────────────────────────────────────────────
    if Cfg.Aimbot.Enabled then
        local target
        if Cfg.Aimbot.TargetLock and Cfg.Aimbot.LockedTarget then
            local c = Cfg.Aimbot.LockedTarget.Character
            target  = c and getAimPart(c, Cfg.Aimbot.TargetPart)
        else
            target = getBestTarget(Cfg.Aimbot.FOV, Cfg.Aimbot.TargetPart)
        end
        if target then
            local aimPos = predictedPos(target)
            local sp     = camera:WorldToViewportPoint(aimPos)
            local center = camera.ViewportSize / 2
            local delta  = Vector2.new(sp.X, sp.Y) - center
            local smooth = math.clamp(1 - Cfg.Aimbot.Smooth, 0.01, 1)
            if mousemoverel_ then
                pcall(mousemoverel_, delta.X * smooth, delta.Y * smooth)
            end
            if Cfg.SilentAim.Manipulate then applyManip(target) end
        end
    end

    -- ── SILENT AIM ───────────────────────────────────────────────
    if Cfg.SilentAim.Enabled then
        local target = getBestTarget(Cfg.SilentAim.FOV, Cfg.Aimbot.TargetPart)
        if target and Cfg.SilentAim.Manipulate then applyManip(target) end
    end

    -- ── TRIGGERBOT ───────────────────────────────────────────────
    if Cfg.TriggerBot.Enabled and (now - RageState.lastTrigger) > Cfg.TriggerBot.Delay then
        local target = getBestTarget(60, Cfg.Aimbot.TargetPart)
        if target then
            if Remotes.UseItem then fireRemote(Remotes.UseItem) end
            RageState.lastTrigger = now
        end
    end

    -- ── RAGEBOT (Kicia-native) ────────────────────────────────────
    if Cfg.Ragebot.Enabled and (now - RageState.lastFire) > Cfg.Ragebot.FireRate then
        local targetPlayer = pickTarget()
        if targetPlayer and targetPlayer.Character then
            local char    = targetPlayer.Character
            local aimPart = getAimPart(char, Cfg.Ragebot.HitboxHead and "Head" or "HumanoidRootPart")

            -- backtrack: ghost to recorded position
            local btCF = btBest(targetPlayer)
            local btPart = char:FindFirstChild("HumanoidRootPart")
            if btCF and btPart then
                -- fire at the backtracked position via cam manip
                local head = lp.Character and lp.Character:FindFirstChild("Head")
                if head and Remotes.CamRot then
                    local dir = (btCF.Position - head.Position).Unit
                    pcall(function()
                        Remotes.CamRot:FireServer(CFrame.new(head.Position, head.Position + dir))
                    end)
                end
            else
                applyManip(aimPart)
            end

            -- desync
            if Cfg.Ragebot.Desync then applyDesync() end

            -- auto weapon switch before firing
            if Cfg.Ragebot.AutoWeapon then
                autoWeaponSwitch(lp.Character)
            end

            -- fire
            if Cfg.Ragebot.AutoFire then
                local shots = Cfg.Ragebot.RapidFire
                    and math.clamp(Cfg.Ragebot.RapidFireCount, 1, 5)
                    or  math.clamp(Cfg.Ragebot.ShootAttempts,  1, 10)
                for _ = 1, shots do
                    if Cfg.Ragebot.Mode == "Gun" or Cfg.Ragebot.Mode == "Auto" then
                        if Remotes.UseItem then fireRemote(Remotes.UseItem) end
                    end
                    if Cfg.Ragebot.Mode == "Melee" or Cfg.Ragebot.Mode == "Auto" then
                        if Remotes.QuickAttack then fireRemote(Remotes.QuickAttack) end
                    end
                end
            end

            -- void spam hide: teleport local char root far then back
            if Cfg.Ragebot.VoidSpamHide then
                local localRoot = getLocalRoot()
                if localRoot then
                    local origCF = localRoot.CFrame
                    pcall(function() localRoot.CFrame = CFrame.new(0, -1e4, 0) end)
                    task.defer(function() pcall(function() localRoot.CFrame = origCF end) end)
                end
            end

            RageState.lastFire = now
        end
    end

    -- ── ESP UPDATE ───────────────────────────────────────────────
    local anyVisual = Cfg.Visuals.ESP or Cfg.Visuals.NameTags
        or Cfg.Visuals.HealthBars or Cfg.Visuals.Snaplines
    if anyVisual then
        updateESP()
    else
        for _, obj in pairs(ESPObjects) do
            for _, o in pairs(obj) do o.Visible = false end
        end
    end

    -- ── FOV CIRCLE ───────────────────────────────────────────────
    if Cfg.Visuals.FOVCircle then
        local r = Cfg.SilentAim.Enabled and Cfg.SilentAim.FOV
            or (Cfg.Aimbot.Enabled and Cfg.Aimbot.FOV or 150)
        fovCircle.Position = camera.ViewportSize / 2
        fovCircle.Radius   = r
        fovCircle.Visible  = true
    else
        fovCircle.Visible = false
    end

    -- ── FLY ──────────────────────────────────────────────────────
    if Cfg.Movement.Fly and flyBody then
        local cf  = camera.CFrame
        local vel = Vector3.zero
        local spd = Cfg.Movement.FlySpeed
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then vel = vel + cf.LookVector  * spd end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then vel = vel - cf.LookVector  * spd end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then vel = vel - cf.RightVector * spd end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then vel = vel + cf.RightVector * spd end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space)       then vel = vel + Vector3.new(0, spd, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then vel = vel - Vector3.new(0, spd, 0) end
        flyBody.bv.Velocity = vel
        flyBody.bg.CFrame   = cf
    end

    -- ── SPINBOT ───────────────────────────────────────────────────
    if Cfg.Misc.SpinBot then
        local localRoot = getLocalRoot()
        if localRoot then
            spinAngle = (spinAngle + Cfg.Misc.SpinSpeed) % 360
            local orig = localRoot.CFrame
            pcall(function()
                localRoot.CFrame = CFrame.new(orig.Position)
                    * CFrame.Angles(0, math.rad(spinAngle), 0)
            end)
        end
    end
end)

        if hum then hum:ChangeState(Enum.HumanoidStateType.Jumping) end
    end
end

    if hum then hum.StateChanged:Connect(onStateChange) end
end
